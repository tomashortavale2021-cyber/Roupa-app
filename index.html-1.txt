  // Calçado (migra calcado -> shoes)

  const shoes=rSh.filter(c=>c&&typeof c.name==='string'&&c.name.trim()).map(c=>({

}

const atLoc=(arr,loc)=>arr.filter(x=>x.location===loc);
function cntSt(arr){let v=0,a=0,r=0;for(const i of arr){if(i.status==='verde')v++;else if(i.status==='amarelo')a++;else r++;}return{v,a,r,u:v+a};}

function computeMetrics(){
  const m=state.metas,D=Math.max(1,m.days);
  const li=atLoc(state.items,'lisboa'),ls=atLoc(state.shoes,'lisboa');
  const bV=state.underwear.boxers.verde,mV=state.underwear.meias.verde;
  const tc=cntSt(li.filter(i=>TSHIRT_CAM_CATS.has(i.category)));
  const cam=cntSt(li.filter(i=>i.category==='Camisolas'));
  const cas=cntSt(li.filter(i=>i.category==='Casacos'));
  const cal=cntSt(li.filter(i=>i.category==='Calças/Calções'));
  const dailySh=ls.filter(s=>!s.isPadel).length,padelSh=ls.filter(s=>s.isPadel).length;

  const pjMissing=[];let pjV=0;
  for(const d of PIJ_DEFS){
    if(!d.required)continue;
    const p=state.pijama.find(x=>x.id===d.id);
    if(p&&p.inLisboa&&p.status==='verde')pjV++;
    else pjMissing.push(`${d.name} (${!p||!p.inLisboa?'em falta':'Vermelho'})`);
  }
  const pjT=state.pijama.find(x=>x.id==='tshirt_pijama');
  const pjOpt=pjT?(pjT.inLisboa?`T-shirt: Lisboa · ${ST_LBL[pjT.status]} (Opcional)`:'T-shirt: Em falta (Opcional)'):'';

  const pillars=[
    {k:'boxers',lbl:'Boxers',kind:'bin',v:bV,a:0,u:bV,min:m.boxers,ideal:m.boxers,days:m.boxers?Math.floor(bV/m.boxers*D):Infinity},
    {k:'meias',lbl:'Meias',kind:'bin',v:mV,a:0,u:mV,min:m.meias,ideal:m.meias,days:m.meias?Math.floor(mV/m.meias*D):Infinity},
    {k:'tops',lbl:'T-shirt/Camisas',kind:'std',v:tc.v,a:tc.a,u:tc.u,min:m.tops,ideal:m.tops,days:m.tops?Math.floor(tc.u/m.tops*D):Infinity,sub:'Camisas + T-Shirts/Polos'},
    {k:'camisolas',lbl:'Camisolas',kind:'mi',v:cam.v,a:cam.a,u:cam.u,min:m.camisolasMin,ideal:m.camisolasIdeal,days:m.camisolasMin?(cam.u>=m.camisolasMin?D:0):Infinity},
    {k:'casacos',lbl:'Casacos',kind:'mi',v:cas.v,a:cas.a,u:cas.u,min:m.casacosMin,ideal:m.casacosIdeal,days:m.casacosMin?(cas.u>=m.casacosMin?D:0):Infinity},
    {k:'calcas',lbl:'Calças/Calções',kind:'std',v:cal.v,a:cal.a,u:cal.u,min:m.calcas,ideal:m.calcas,days:m.calcas?Math.floor(cal.u/m.calcas*D):Infinity},
    {k:'sapatilhas',lbl:'Sapatilhas / Calçado útil',kind:'sh',v:dailySh,a:0,u:dailySh,min:m.sapatilhasMin,ideal:m.sapatilhasIdeal,days:m.sapatilhasMin?(dailySh>=m.sapatilhasMin?D:0):Infinity,padel:padelSh,onlyPadel:m.sapatilhasMin>0&&dailySh===0&&padelSh>0},
    {k:'pijama',lbl:'Pijama',kind:'pj',v:pjV,a:0,u:pjV,min:3,ideal:3,days:pjV===3?D:0,missing:pjMissing,optTxt:pjOpt}
  ];

  const active=pillars.filter(p=>p.min>0);
  const rawMin=active.length?Math.min(...active.map(p=>p.days)):D;
  const days=rawMin===Infinity?D:rawMin;
  const minCovered=days>=D;
  const bottlenecks=minCovered?[]:active.filter(p=>p.days===days).map(p=>p.lbl);

  const idealGaps=[];
  const gCam=Math.max(0,m.camisolasIdeal-cam.u),gCas=Math.max(0,m.casacosIdeal-cas.u),gSh=Math.max(0,m.sapatilhasIdeal-dailySh);
  if(gCam>0)idealGaps.push(`+${gCam} camisola${gCam>1?'s':''}`);
  if(gCas>0)idealGaps.push(`+${gCas} casaco${gCas>1?'s':''}`);
  if(gSh>0)idealGaps.push(`+${gSh} sapatilha${gSh>1?'s':''}`);
  const allIdeal=minCovered&&idealGaps.length===0;

  const minDef={
    boxers:Math.max(0,m.boxers-bV),meias:Math.max(0,m.meias-mV),
    tops:Math.max(0,m.tops-tc.u),camisolas:Math.max(0,m.camisolasMin-cam.u),
    casacos:Math.max(0,m.casacosMin-cas.u),calcas:Math.max(0,m.calcas-cal.u),
    sapatilhas:Math.max(0,m.sapatilhasMin-dailySh),pijama:pjMissing
  };
  return{D,days,minCovered,allIdeal,bottlenecks,idealGaps,pillars,minDef};
}

// Render: Autonomia (8 Pilares)
function renderAutonomy(){
  const m=computeMetrics(),D=m.D;
  const lvl=m.minCovered?'g':(m.days>=Math.ceil(D/2)?'y':'r');
  let badge='',banner='';
  if(!m.minCovered){
    badge=`<span class="pill r">${m.days} / ${D} dias</span>`;
    const ex=m.pillars[6].onlyPadel?' • Só tens calçado de padel em Lisboa (não conta).':'';
    banner=`<div class="alert-box">⚠️ Gargalo${m.bottlenecks.length>1?'s':''}: ${esc(m.bottlenecks.join(', '))}${ex}</div>`;
  }else if(m.allIdeal){
    badge=`<span class="pill a">★ Autonomia Ideal (${D}d)</span>`;
  }else{
    badge=`<span class="pill g">✓ Mínimo ${D}d coberto</span> <span class="pill a">Falta p/ ideal: ${esc(m.idealGaps.join(', '))}</span>`;
    banner=`<div class="alert-box ideal">✨ Falta p/ ideal: ${esc(m.idealGaps.join(', '))}</div>`;
  }

  const pHtml=m.pillars.map(p=>{
    const bot=!m.minCovered&&p.min>0&&p.days===m.days;
    const isStar=p.min>0&&p.u>=p.ideal&&(p.kind==='mi'||p.kind==='sh');
    const isOk=p.min>0&&p.days>=D;
    const cls='pbox'+(bot?' bad':(isStar?' star':(isOk?' ok':'')));
    let tag='';
    if(p.min===0)tag='<span class="pday">Ignorado</span>';
    else if(p.kind==='mi'||p.kind==='sh'||p.kind==='pj'){
      tag=isStar?'<span class="pday star">★ Ideal</span>':(isOk?`<span class="pday ok">✓ ${D}d</span>`:'<span class="pday bad">0d</span>');
    }else tag=`<span class="pday ${isOk?'ok':(bot?'bad':'')}">${p.days}d</span>`;

    let val='',sub='';
    if(p.kind==='mi'){
      const dec=p.a>0?` (${p.v}V + ${p.a}A)`:'';
      val=`${p.u} · Mín ${p.min} / Ideal ${p.ideal}`;
      sub=`Verde + Amarelo${dec}`;
    }else if(p.kind==='sh'){
      val=`${p.u} · Mín ${p.min} / Ideal ${p.ideal}`;
      sub=p.onlyPadel?'<span class="psub bad">Só tens calçado de padel em Lisboa (não conta)</span>':`Uso diário${p.padel?` · +${p.padel} Padel`:''}`;
    }else if(p.kind==='pj'){
      val=`${p.u}/3 obrig.`;
      sub=p.missing.length?`<span class="psub bad">Falta: ${esc(p.missing.join(', '))}</span>`:'3/3 a Verde em Lisboa';
      if(p.optTxt)sub+=`<div class="psub">${esc(p.optTxt)}</div>`;
    }else{
      const dec=p.kind==='std'&&p.a>0?` <span style="font-size:.75rem;color:var(--tx2)">(${p.v}V + ${p.a}A)</span>`:'';
      val=`${p.u} / ${p.min}${dec}`;
      sub=p.kind==='bin'?'Só Verde (Limpo)':(p.sub||'Verde + Amarelo');
    }

    const maxB=Math.max(p.ideal,p.min,1);
    const gW=Math.min(100,p.v/maxB*100),yW=Math.min(100-gW,p.a/maxB*100);
    const mark=(p.kind==='mi'||p.kind==='sh')&&p.min>0&&p.ideal>p.min?`<div class="bar-mark" style="left:${(p.min/maxB*100).toFixed(1)}%" title="Mínimo ${p.min}"></div>`:'';
    return `<div class="${cls}">
      <div class="ptop"><span class="plbl">${esc(p.lbl)}</span>${tag}</div>
      <div class="pval">${val}</div>
      <div class="bar-wrap"><div class="bar-inner"><div class="bar-g" style="width:${gW.toFixed(1)}%"></div>${yW>0?`<div class="bar-y" style="width:${yW.toFixed(1)}%"></div>`:''}</div>${mark}</div>
      <div class="psub">${sub}</div>
    </div>`;
  }).join('');

  $('autonomy-box').innerHTML=`<div class="auto-card">
    <div class="auto-main">
      <div class="auto-box ${lvl}"><span class="auto-num">${m.days}/${D}</span><span class="auto-unit">dias</span></div>
      <div>
        <div style="font-weight:850;font-size:1.03rem">Autonomia em Lisboa: ${m.days} / ${D} dias</div>
        <div class="auto-badges">${badge}</div>
      </div>
    </div>
    ${banner}
    <div class="pillars">${pHtml}</div>
  </div>`;
}

// Render: Mala (Lisboa <-> Coimbra)
// Regra 1: Em Lisboa -> Coimbra, Amarelas, Verdes e Pijama Verde são opcionais (desmarcados por defeito). Roupa interior só Vermelho.
// Regra 2: Em Coimbra -> Lisboa, NENHUMA peça de pijama é pré-selecionada automaticamente.
function syncMala(){
  const li=atLoc(state.items,'lisboa'),ci=atLoc(state.items,'casa');
  const lIds=new Set(li.map(i=>i.id)),cIds=new Set(ci.map(i=>i.id));
  const lsIds=new Set(atLoc(state.shoes,'lisboa').map(s=>s.id)),csIds=new Set(atLoc(state.shoes,'casa').map(s=>s.id));
  for(const id of [...ui.mRet])if(!lIds.has(id))ui.mRet.delete(id);
  for(const id of [...ui.mRetSync])if(!lIds.has(id))ui.mRetSync.delete(id);
  for(const id of [...ui.mGo])if(!cIds.has(id))ui.mGo.delete(id);
  for(const id of [...ui.mRetSh])if(!lsIds.has(id))ui.mRetSh.delete(id);
  for(const id of [...ui.mGoSh])if(!csIds.has(id))ui.mGoSh.delete(id);

  // Lisboa -> Coimbra: pré-seleciona automaticamente peças a Vermelho; mantém seleções manuais de Amarelas/Verdes
  li.forEach(i=>{
    if(i.status==='vermelho'){if(!ui.mRetSync.has(i.id)){ui.mRet.add(i.id);ui.mRetSync.add(i.id);}}
    else ui.mRetSync.delete(i.id);
  });

  for(const d of PIJ_DEFS){
    const p=state.pijama.find(x=>x.id===d.id);
    if(!p.inLisboa){ui.mRetPij.delete(d.id);ui.mRetPijSync.delete(d.id);}
    else if(p.status==='vermelho'){if(!ui.mRetPijSync.has(d.id)){ui.mRetPij.add(d.id);ui.mRetPijSync.add(d.id);}}
    else ui.mRetPijSync.delete(d.id); // Se Verde em Lisboa, preserva eventual seleção manual em ui.mRetPij

    // Coimbra -> Lisboa: nunca pré-selecionar pijama; remove apenas se já estiver Verde em Lisboa
    if(p.inLisboa&&p.status==='verde')ui.mGoPij.delete(d.id);
  }

  const mb=state.underwear.boxers.vermelho,mm=state.underwear.meias.vermelho;
  ui.mRetB=ui.mRetB===null?mb:Math.min(mb,Math.max(0,ui.mRetB));
  ui.mRetM=ui.mRetM===null?mm:Math.min(mm,Math.max(0,ui.mRetM));
  const m=computeMetrics();
  if(ui.mGoB===null)ui.mGoB=m.minDef.boxers;
  if(ui.mGoM===null)ui.mGoM=m.minDef.meias;
}

  }).join('\n\n');

function render(){
  const lCnt=atLoc(state.items,'lisboa').length+atLoc(state.shoes,'lisboa').length+state.pijama.filter(p=>p.inLisboa).length;
  const cCnt=atLoc(state.items,'casa').length+atLoc(state.shoes,'casa').length+4;

    +(state.underwear.boxers.vermelho?1:0)+(state.underwear.meias.vermelho?1:0);

  ['bd-m','bm-m'].forEach(id=>{$(id).textContent=String(alerts);$(id).classList.toggle('hidden',!alerts);});

  document.querySelectorAll('[data-action="switch-tab"]').forEach(b=>b.classList.toggle('active',b.dataset.tab===ui.tab));

    if(id){
      const it=state.items.find(x=>x.id===id);
      if(it){it.name=name;it.category=normCat($('fi-cat').value,name);it.location=loc;it.note=note;it.status=st;it.reasons=rs;log(`Editada "${it.name}".`);}
    }else{
      const ni={id:uid('it'),name,category:normCat($('fi-cat').value,name),location:loc,status:st,reasons:rs,uses:0,note};
      state.items.unshift(ni);if(loc==='lisboa'&&st==='vermelho'){ui.mRet.add(ni.id);ui.mRetSync.add(ni.id);}
      log(`Adicionada "${name}" (${ni.category}).`);
    }
    save();closeModal('m-item');render();toast('Peça guardada!');
  }else if(a==='sub-shoe'){
    const id=$('fs-id').value,name=$('fs-name').value.trim(),loc=$('fs-loc').value==='casa'?'casa':'lisboa',isPadel=$('fs-padel').checked,note=$('fs-note').value.trim();
    if(!name){toast('Indica o nome do calçado.','err');return;}
    if(id){
      const s=state.shoes.find(x=>x.id===id);
      if(s){s.name=name;s.location=loc;s.isPadel=isPadel;s.note=note;log(`Editado calçado "${name}".`);}
    }else{
      state.shoes.unshift({id:uid('sh'),name,location:loc,isPadel,note});log(`Adicionado calçado "${name}".`);
    }
    save();closeModal('m-shoe');render();toast('Calçado guardado!');
  }else if(a==='sub-metas'){
    const fd=new FormData(f),g=(k,mn,mx,fb)=>clamp(fd.get(k),mn,mx,fb);
    const nm={
      days:g('days',1,7,4),boxers:g('boxers',0,30,4),meias:g('meias',0,30,4),tops:g('tops',0,30,4),
      camisolasMin:g('camisolasMin',0,15,1),camisolasIdeal:g('camisolasIdeal',0,20,3),
      casacosMin:g('casacosMin',0,10,1),casacosIdeal:g('casacosIdeal',0,15,2),
      calcas:g('calcas',0,15,2),sapatilhasMin:g('sapatilhasMin',0,10,1),sapatilhasIdeal:g('sapatilhasIdeal',0,10,2)
    };
    if(nm.camisolasIdeal<nm.camisolasMin)nm.camisolasIdeal=nm.camisolasMin;
    if(nm.casacosIdeal<nm.casacosMin)nm.casacosIdeal=nm.casacosMin;
    if(nm.sapatilhasIdeal<nm.sapatilhasMin)nm.sapatilhasIdeal=nm.sapatilhasMin;
    state.metas=nm;ui.mGoB=ui.mGoM=null;log('Metas semanais guardadas.');save();render();toast('Metas guardadas!');
  }
});
