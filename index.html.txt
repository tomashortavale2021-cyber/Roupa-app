      casacosMin: Math.max(1, num(g.casacosMin != null ? g.casacosMin : g.casacos, dg.casacosMin)),
      casacosIdeal: Math.max(1, num(g.casacosIdeal, dg.casacosIdeal)),
      sapatilhasMin: Math.max(1, num(g.sapatilhasMin != null ? g.sapatilhasMin : g.sapatilhas, dg.sapatilhasMin)),
      sapatilhasIdeal: Math.max(1, num(g.sapatilhasIdeal, dg.sapatilhasIdeal))
    };
    if(out.goals.camisolasIdeal < out.goals.camisolasMin) out.goals.camisolasIdeal = out.goals.camisolasMin;
    if(out.goals.casacosIdeal < out.goals.casacosMin) out.goals.casacosIdeal = out.goals.casacosMin;
    if(out.goals.sapatilhasIdeal < out.goals.sapatilhasMin) out.goals.sapatilhasIdeal = out.goals.sapatilhasMin;

    // Underwear (strictly binary verde / vermelho)
    const uw = raw.underwear || raw.roupaInterior || {};
    ['boxers','meias'].forEach(function(k){
      const src = uw[k] || {};
      const v = num(src.verde != null ? src.verde : src.green, 0);
      const a = num(src.amarelo != null ? src.amarelo : src.yellow, 0);
      const r = num(src.vermelho != null ? src.vermelho : src.red, 0);
      out.underwear[k] = { verde: v, vermelho: r + a };
    });

    // Pijama
    const rp = raw.pijama || {};
    PIJAMA_DEFS.forEach(function(d){
      const src = rp[d.key];
      if(src && typeof src === 'object'){
        const st = normBinary(src.status);
        out.pijama[d.key] = {
          inLisboa: src.inLisboa !== false,
          status: st,
          reasons: st === 'vermelho' && Array.isArray(src.reasons) ? src.reasons.filter(function(r){ return REASONS_PIJAMA.indexOf(r) !== -1; }) : [],
          uses: num(src.uses, 0)
        };
      }
    });

    // Items (and migrate old fixed pijama items if present)
    const rawItems = Array.isArray(raw.items) ? raw.items : (Array.isArray(raw.pecas) ? raw.pecas : []);
    rawItems.forEach(function(it){
      if(!it || !it.name) return;
      const pk = (it.category === 'Pijama' || it.fixedId) ? matchPijamaKey(it) : null;
      if(pk && !rp[pk]){
        const st = normBinary(it.status);
        out.pijama[pk] = {
          inLisboa: (it.location || 'lisboa') === 'lisboa',
          status: st,
          reasons: st === 'vermelho' && Array.isArray(it.reasons) ? it.reasons.filter(function(r){ return REASONS_PIJAMA.indexOf(r) !== -1; }) : [],
          uses: num(it.uses, 0)
        };
        return;
      }
      const st = normStatus(it.status);
      out.items.push({
        id: it.id || uid('it'),
        name: String(it.name).trim(),
        category: classifyCategory(it.category, it.name),
        location: it.location === 'casa' ? 'casa' : 'lisboa',
        status: st,
        reasons: st === 'verde' ? [] : (Array.isArray(it.reasons) ? it.reasons.filter(function(r){ return REASONS_ITEM.indexOf(r) !== -1; }) : []),
        uses: num(it.uses, 0),
        note: String(it.note || '').trim()
      });
    });

    // Shoes
    const rawShoes = Array.isArray(raw.shoes) ? raw.shoes : (Array.isArray(raw.calcado) ? raw.calcado : []);
    rawShoes.forEach(function(sh){
      if(!sh || !sh.name) return;
      const nm = String(sh.name).trim();
      const isPadel = typeof sh.isPadel === 'boolean' ? sh.isPadel : (/padel/i.test(nm));
      out.shoes.push({
        id: sh.id || uid('sh'),
        name: nm,
        location: sh.location === 'casa' ? 'casa' : 'lisboa',
        isPadel: isPadel,
        note: String(sh.note || '').trim()
      });
    });

    // History
    if(Array.isArray(raw.history)){
      out.history = raw.history.slice(0, 15).map(function(h){
        if(typeof h === 'string') return { time:'', text:h };
        return { time: String(h.time || ''), text: String(h.text || '') };
      });
    }
    return out;
  }

  // Autonomy calculation (8 pillars)
  function calcAutonomy(){
    const g = state.goals;
    const lisboaItems = state.items.filter(function(i){ return i.location === 'lisboa'; });
    const countCat = function(cats){
      let v = 0, a = 0;
      lisboaItems.forEach(function(i){
        if(cats.indexOf(i.category) !== -1){
          if(i.status === 'verde') v++;
          else if(i.status === 'amarelo') a++;
        }
      });
      return { v:v, a:a, tot:v+a };
    };

    const tops = countCat(['Camisas','T-Shirts/Polos']);
    const camisolas = countCat(['Camisolas']);
    const casacos = countCat(['Casacos']);
    const calcas = countCat(['Calças/Calções']);
    const shoesLisboa = state.shoes.filter(function(s){ return s.location === 'lisboa'; });
    const usefulShoes = shoesLisboa.filter(function(s){ return !s.isPadel; }).length;
    const padelShoes = shoesLisboa.filter(function(s){ return s.isPadel; }).length;

    const pijamaMissing = [];
    let pijamaReqOk = 0;
    PIJAMA_DEFS.forEach(function(d){
      if(!d.required) return;
      const st = state.pijama[d.key];
      if(st && st.inLisboa && st.status === 'verde') pijamaReqOk++;
      else pijamaMissing.push(d.name + (!st || !st.inLisboa ? ' (em falta)' : ' (Vermelho)'));
    });

    const daysForLinear = function(have, target){
      if(target <= 0) return g.days;
      if(have >= target) return g.days;
      return Math.floor((have / target) * g.days);
    };

    const dBoxers = daysForLinear(state.underwear.boxers.verde, g.boxers);
    const dMeias = daysForLinear(state.underwear.meias.verde, g.meias);
    const dTops = daysForLinear(tops.tot, g.tops);
    const dCalcas = daysForLinear(calcas.tot, g.calcas);
    const dCamisolas = camisolas.tot >= g.camisolasMin ? g.days : 0;
    const dCasacos = casacos.tot >= g.casacosMin ? g.days : 0;
    const dShoes = usefulShoes >= g.sapatilhasMin ? g.days : 0;
    const dPijama = pijamaReqOk === 3 ? g.days : 0;

    const realDays = Math.min(dBoxers, dMeias, dTops, dCalcas, dCamisolas, dCasacos, dShoes, dPijama);

    const bottlenecks = [];
    if(state.underwear.boxers.verde < g.boxers) bottlenecks.push('Boxers ('+state.underwear.boxers.verde+'/'+g.boxers+')');
    if(state.underwear.meias.verde < g.meias) bottlenecks.push('Meias ('+state.underwear.meias.verde+'/'+g.meias+')');
    if(tops.tot < g.tops) bottlenecks.push('T-shirt/Camisas ('+tops.tot+'/'+g.tops+')');
    if(camisolas.tot < g.camisolasMin) bottlenecks.push('Camisolas ('+camisolas.tot+'/'+g.camisolasMin+' mín)');
    if(casacos.tot < g.casacosMin) bottlenecks.push('Casacos ('+casacos.tot+'/'+g.casacosMin+' mín)');
    if(calcas.tot < g.calcas) bottlenecks.push('Calças/Calções ('+calcas.tot+'/'+g.calcas+')');
    if(usefulShoes < g.sapatilhasMin){
      bottlenecks.push(padelShoes > 0 ? 'Sapatilhas (só tens padel em Lisboa)' : 'Sapatilhas ('+usefulShoes+'/'+g.sapatilhasMin+' mín)');
    }
    if(pijamaReqOk < 3) bottlenecks.push('Pijama ('+pijamaReqOk+'/3 obrig.)');

    const idealGaps = [];
    if(camisolas.tot < g.camisolasIdeal) idealGaps.push('+'+(g.camisolasIdeal - camisolas.tot)+' camisola(s)');
    if(casacos.tot < g.casacosIdeal) idealGaps.push('+'+(g.casacosIdeal - casacos.tot)+' casaco(s)');
    if(usefulShoes < g.sapatilhasIdeal) idealGaps.push('+'+(g.sapatilhasIdeal - usefulShoes)+' sapatilha(s)');

    return {
      realDays: realDays,
      targetDays: g.days,
      minCovered: bottlenecks.length === 0,
      idealCovered: bottlenecks.length === 0 && idealGaps.length === 0,
      bottlenecks: bottlenecks,
      idealGaps: idealGaps,
      boxers: state.underwear.boxers.verde,
      meias: state.underwear.meias.verde,
      tops: tops,
      camisolas: camisolas,
      casacos: casacos,
      calcas: calcas,
      usefulShoes: usefulShoes,
      padelShoes: padelShoes,
      pijamaReqOk: pijamaReqOk,
      pijamaMissing: pijamaMissing,
      pijamaTshirtOk: !!(state.pijama.tshirt_pijama && state.pijama.tshirt_pijama.inLisboa && state.pijama.tshirt_pijama.status === 'verde')
    };
  }

  function renderPillarBar(v, a, maxVal, minVal, isBlue){
    const denom = Math.max(1, maxVal);
    const gv = Math.min(v, denom);
    const av = Math.min(a, Math.max(0, denom - gv));
    const pG = Math.round((gv / denom) * 100);
    const pA = Math.round((av / denom) * 100);
    const marker = (minVal && minVal < maxVal)
      ? '<span class="bar-marker" style="left:'+Math.round((minVal/denom)*100)+'%" title="Mínimo '+minVal+'"></span>'
      : '';
    return '<div class="bar-wrap">' +
      (pG > 0 ? '<div class="'+(isBlue ? 'bar-b' : 'bar-g')+'" style="width:'+pG+'%"></div>' : '') +
      (pA > 0 ? '<div class="bar-a" style="width:'+pA+'%"></div>' : '') +
      marker + '</div>';
  }

  function renderLisboa(){
    const q = ui.search.trim().toLowerCase();
    const allLisboa = state.items.filter(function(i){ return i.location === 'lisboa'; });
    const filtered = allLisboa.filter(function(i){
      if(ui.statusFilter !== 'all' && i.status !== ui.statusFilter) return false;
      if(ui.catFilter !== 'all' && i.category !== ui.catFilter) return false;
      if(q && i.name.toLowerCase().indexOf(q) === -1 && i.note.toLowerCase().indexOf(q) === -1) return false;
      return true;
    });

    const catsToShow = ui.catFilter === 'all' ? CATEGORIES : [ui.catFilter];
    const catsHtml = catsToShow.map(function(cat){
      const arr = filtered.filter(function(i){ return i.category === cat; });
      const allCat = allLisboa.filter(function(i){ return i.category === cat; });
      const v = allCat.filter(function(i){ return i.status === 'verde'; }).length;
      const am = allCat.filter(function(i){ return i.status === 'amarelo'; }).length;
      const r = allCat.filter(function(i){ return i.status === 'vermelho'; }).length;
      return '<div class="card">' +
        '<div class="cat-head">' +
          '<div class="cat-title">' +
            '<span>'+esc(cat)+' ('+arr.length+')</span>' +
            (v ? '<span class="pill green">'+v+'V</span>' : '') +
            (am ? '<span class="pill amber">'+am+'A</span>' : '') +
            (r ? '<span class="pill red">'+r+'R</span>' : '') +
          '</div>' +
          '<button type="button" class="xs" data-action="open-add-item" data-cat="'+esc(cat)+'" data-loc="lisboa">+ Adicionar</button>' +
        '</div>' +
        (arr.length ? arr.map(function(it){ return renderItemCard(it, false); }).join('') : '<div class="empty">Sem peças nesta categoria'+(q || ui.statusFilter !== 'all' ? ' com os filtros atuais' : '')+'.</div>') +
      '</div>';
    }).join('');

    const filtersCard = '<section class="card">' +
      '<div class="card-head"><h2 class="card-title">👔 Peças em Lisboa por Categoria</h2><span class="pill gray">'+allLisboa.length+' peças</span></div>' +
      '<div class="search-bar">' +
        '<input type="text" id="search-input" placeholder="Pesquisar peça ou nota..." value="'+esc(ui.search)+'">' +
        '<select id="cat-filter" style="width:auto;min-width:150px">' +
          '<option value="all"'+(ui.catFilter==='all'?' selected':'')+'>Todas as categorias</option>' +
          CATEGORIES.map(function(c){ return '<option value="'+esc(c)+'"'+(ui.catFilter===c?' selected':'')+'>'+esc(c)+'</option>'; }).join('') +
        '</select>' +
      '</div>' +
      '<div class="chips">' +
        [['all','Todos'],['verde','Verde'],['amarelo','Amarelo'],['vermelho','Vermelho']].map(function(pair){
          return '<button type="button" class="chip'+(ui.statusFilter===pair[0]?' active':'')+'" data-action="filter-st" data-st="'+pair[0]+'">'+pair[1]+'</button>';
        }).join('') +
      '</div>' +
    '</section>';

    return renderAutonomyCard() + renderUnderwearCard() + renderPijamaLisboaCard() + filtersCard + catsHtml + renderShoesCard('lisboa');
  }

  function buildMarkdownSummary(){
    const a = calcAutonomy();
    const g = state.goals;
    const lines = [];
    lines.push('# Guarda-Roupa Lisboa — Ponto de Situação ('+nowStr()+')');
    lines.push('');
    lines.push('## Autonomia em Lisboa (Meta: '+g.days+' dias Seg–Qui)');
    lines.push('- **Estado geral:** '+a.realDays+'/'+g.days+' dias'+(a.idealCovered ? ' (Autonomia Ideal ★)' : (a.minCovered ? ' (Mínimo coberto ✓; falta p/ ideal: '+a.idealGaps.join(', ')+')' : ' (Gargalos: '+a.bottlenecks.join(', ')+')')));
    lines.push('- **Boxers (só Verde):** '+state.underwear.boxers.verde+'V / '+state.underwear.boxers.vermelho+'R (meta '+g.boxers+')');
    lines.push('- **Meias (só Verde):** '+state.underwear.meias.verde+'V / '+state.underwear.meias.vermelho+'R (meta '+g.meias+')');
    lines.push('- **T-shirt/Camisas:** '+a.tops.tot+'/'+g.tops+fmtVA(a.tops));
    lines.push('- **Camisolas:** '+a.camisolas.tot+' (Mín '+g.camisolasMin+' / Ideal '+g.camisolasIdeal+')'+fmtVA(a.camisolas));
    lines.push('- **Casacos:** '+a.casacos.tot+' (Mín '+g.casacosMin+' / Ideal '+g.casacosIdeal+')'+fmtVA(a.casacos));
    lines.push('- **Calças/Calções:** '+a.calcas.tot+'/'+g.calcas+fmtVA(a.calcas));
    lines.push('- **Sapatilhas úteis (não-Padel):** '+a.usefulShoes+' (Mín '+g.sapatilhasMin+' / Ideal '+g.sapatilhasIdeal+')'+(a.padelShoes ? ' · +'+a.padelShoes+' de Padel' : ''));
    lines.push('- **Pijama (obrigatórias a Verde):** '+a.pijamaReqOk+'/3'+(a.pijamaMissing.length ? ' (Em falta: '+a.pijamaMissing.join(', ')+')' : ' ✓'));
    lines.push('');
    lines.push('## Pijama');
    PIJAMA_DEFS.forEach(function(d){
      const p = state.pijama[d.key];
      const tag = d.required ? 'Obrigatória' : 'Opcional';
      if(!p.inLisboa) lines.push('- '+d.name+' ('+tag+'): **Em falta em Lisboa** (reserva em Coimbra)');
      else lines.push('- '+d.name+' ('+tag+'): **'+p.status.toUpperCase()+'** (usos: '+p.uses+(p.reasons.length ? '; '+p.reasons.join(', ') : '')+')');
    });

    const addLoc = function(loc, title){
      lines.push('');
      lines.push('## '+title);
      CATEGORIES.forEach(function(cat){
        const arr = state.items.filter(function(i){ return i.location === loc && i.category === cat; });
        lines.push('### '+cat+' ('+arr.length+')');
        if(!arr.length) lines.push('- (vazio)');
        else arr.forEach(function(it){
          const extra = [it.status.toUpperCase(), 'usos: '+it.uses, it.reasons.join(', '), it.note].filter(Boolean).join(' · ');
          lines.push('- '+it.name+' — '+extra);
        });
      });
      const sh = state.shoes.filter(function(s){ return s.location === loc; });
      lines.push('### Calçado ('+sh.length+')');
      if(!sh.length) lines.push('- (vazio)');
      else sh.forEach(function(s){
        lines.push('- '+s.name+' — '+(s.isPadel ? 'Padel (não conta)' : 'Uso diário')+(s.note ? ' · '+s.note : ''));
      });
    };
    addLoc('lisboa', 'Em Lisboa');
    addLoc('casa', 'Em Casa (Coimbra)');
    return lines.join('\n');
  }

  function copyWithFallback(text, title){
    if(navigator.clipboard && navigator.clipboard.writeText){
      navigator.clipboard.writeText(text).then(function(){
        showToast(title + ' copiado!');
      }).catch(function(){
        openCopyModal(text, title);
      });
    } else {
      openCopyModal(text, title);
    }
  }

  // Event delegation: clicks
  document.addEventListener('click', function(e){
    const back = document.getElementById('modal-backdrop');
    if(e.target === back){ closeModal(); return; }
    const btn = e.target.closest('[data-action]');
    if(!btn) return;
    const act = btn.dataset.action;

    if(act === 'tab'){
      ui.tab = btn.dataset.tab;
      if(ui.tab === 'mala') ui.l2cInitialized = false;
      render();
      window.scrollTo(0, 0);
      return;
    }
    if(act === 'close-modal'){ closeModal(); return; }
    if(act === 'select-copy'){
      const ta = document.getElementById('copy-area');
      if(ta){ ta.focus(); ta.select(); try { document.execCommand('copy'); showToast('Copiado!'); } catch(_){} }
      return;
    }

    // Underwear
    if(act === 'uw-adj'){
      const k = btn.dataset.key, st = btn.dataset.st, d = parseInt(btn.dataset.delta, 10);
      state.underwear[k][st] = Math.max(0, state.underwear[k][st] + d);
      ui.l2cInitialized = false;
      saveState(); render(); return;
    }
    if(act === 'uw-use'){
      const k = btn.dataset.key;
      if(state.underwear[k].verde > 0){
        state.underwear[k].verde--;
        state.underwear[k].vermelho++;
        addHistory('Usado 1 '+k+' (Verde → Vermelho)');
        ui.l2cInitialized = false;
        saveState(); render(); showToast('Registado uso de '+k);
      }
      return;
    }

    // Pijama
    if(act === 'pij-status'){
      const k = btn.dataset.key, st = btn.dataset.st;
      state.pijama[k].status = st;
      if(st === 'verde') state.pijama[k].reasons = [];
      ui.l2cInitialized = false;
      saveState(); render(); return;
    }
    if(act === 'pij-reason'){
      const k = btn.dataset.key, r = btn.dataset.reason;
      const arr = state.pijama[k].reasons;
      const idx = arr.indexOf(r);
      if(idx === -1) arr.push(r); else arr.splice(idx, 1);
      saveState(); render(); return;
    }
    if(act === 'pij-use'){
      state.pijama[btn.dataset.key].uses++;
      saveState(); render(); return;
    }
    if(act === 'pij-reset'){
      state.pijama[btn.dataset.key].uses = 0;
      saveState(); render(); return;
    }
    if(act === 'pij-swap'){
      const k = btn.dataset.key;
      const def = PIJAMA_DEFS.find(function(x){ return x.key === k; });
      state.pijama[k] = { inLisboa:true, status:'verde', reasons:[], uses:0 };
      addHistory('Pijama trocado com reserva de Coimbra: '+(def?def.name:k));
      ui.l2cInitialized = false;
      saveState(); render(); showToast('Trocado com reserva limpa!'); return;
    }
    if(act === 'pij-bring'){
      const k = btn.dataset.key;
      const def = PIJAMA_DEFS.find(function(x){ return x.key === k; });
      state.pijama[k] = { inLisboa:true, status:'verde', reasons:[], uses:0 };
      addHistory('Reserva de pijama enviada p/ Lisboa: '+(def?def.name:k));
      ui.l2cInitialized = false;
      saveState(); render(); showToast('Enviado para Lisboa (Verde)'); return;
    }
    if(act === 'pij-to-casa'){
      const k = btn.dataset.key;
      const def = PIJAMA_DEFS.find(function(x){ return x.key === k; });
      state.pijama[k] = { inLisboa:false, status:'verde', reasons:[], uses:0 };
      addHistory('Pijama levado para Coimbra: '+(def?def.name:k));
      ui.l2cInitialized = false;
      saveState(); render(); showToast('Levado para Coimbra'); return;
    }

    // Items
    if(act === 'open-add-item'){ openItemModal(null, btn.dataset.cat, btn.dataset.loc); return; }
    if(act === 'edit-item'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it) openItemModal(it);
      return;
    }
    if(act === 'del-item'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it) openConfirmModal('Eliminar peça', 'Queres eliminar "'+it.name+'" permanentemente?', 'confirm-del-item', it.id);
      return;
    }
    if(act === 'confirm-del-item'){
      const idx = state.items.findIndex(function(x){ return x.id === btn.dataset.id; });
      if(idx !== -1){
        const nm = state.items[idx].name;
        state.items.splice(idx, 1);
        addHistory('Peça eliminada: '+nm);
        ui.l2cInitialized = false;
        saveState(); closeModal(); render(); showToast('Peça eliminada');
      }
      return;
    }
    if(act === 'item-status'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){
        it.status = btn.dataset.st;
        if(it.status === 'verde') it.reasons = [];
        ui.l2cInitialized = false;
        saveState(); render();
      }
      return;
    }
    if(act === 'item-reason'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){
        const r = btn.dataset.reason;
        const i = it.reasons.indexOf(r);
        if(i === -1) it.reasons.push(r); else it.reasons.splice(i, 1);
        saveState(); render();
      }
      return;
    }
    if(act === 'item-use'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){ it.uses++; saveState(); render(); }
      return;
    }
    if(act === 'item-reset'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){ it.uses = 0; saveState(); render(); }
      return;
    }
    if(act === 'item-clean'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){
        it.status = 'verde'; it.reasons = []; it.uses = 0;
        addHistory('Peça pronta em Coimbra: '+it.name);
        saveState(); render(); showToast('Marcada a Verde');
      }
      return;
    }
    if(act === 'item-move'){
      const it = state.items.find(function(x){ return x.id === btn.dataset.id; });
      if(it){
        const loc = btn.dataset.loc;
        it.location = loc;
        if(loc === 'lisboa'){ it.status = 'verde'; it.reasons = []; it.uses = 0; }
        addHistory('"'+it.name+'" → '+(loc==='lisboa'?'Lisboa (Verde)':'Coimbra'));
        ui.l2cInitialized = false;
        saveState(); render(); showToast('Movida para '+(loc==='lisboa'?'Lisboa':'Coimbra'));
      }
      return;
    }

    // Modal status / reasons
    if(act === 'modal-st'){
      ui.modalStatus = btn.dataset.st;
      if(ui.modalStatus === 'verde') ui.modalReasons = [];
      const seg = document.getElementById('m-seg');
      if(seg){
        seg.querySelectorAll('button').forEach(function(b){
          b.className = b.dataset.st === ui.modalStatus ? 'act-'+ui.modalStatus : '';
        });
      }
      const rw = document.getElementById('m-reasons-wrap');
      if(rw) rw.style.display = ui.modalStatus === 'verde' ? 'none' : 'block';
      return;
    }
    if(act === 'modal-reason'){
      const r = btn.dataset.reason;
      const i = ui.modalReasons.indexOf(r);
      if(i === -1) ui.modalReasons.push(r); else ui.modalReasons.splice(i, 1);
      btn.classList.toggle('active', i === -1);
      return;
    }

    // Filters
    if(act === 'filter-st'){ ui.statusFilter = btn.dataset.st; render(); return; }

    // Shoes
    if(act === 'open-add-shoe'){ openShoeModal(null, btn.dataset.loc); return; }
    if(act === 'edit-shoe'){
      const sh = state.shoes.find(function(x){ return x.id === btn.dataset.id; });
      if(sh) openShoeModal(sh);
      return;
    }
    if(act === 'del-shoe'){
      const sh = state.shoes.find(function(x){ return x.id === btn.dataset.id; });
      if(sh) openConfirmModal('Eliminar calçado', 'Queres eliminar "'+sh.name+'"?', 'confirm-del-shoe', sh.id);
      return;
    }
    if(act === 'confirm-del-shoe'){
      const idx = state.shoes.findIndex(function(x){ return x.id === btn.dataset.id; });
      if(idx !== -1){
        const nm = state.shoes[idx].name;
        state.shoes.splice(idx, 1);
        addHistory('Calçado eliminado: '+nm);
        saveState(); closeModal(); render(); showToast('Calçado eliminado');
      }
      return;
    }
    if(act === 'shoe-padel'){
      const sh = state.shoes.find(function(x){ return x.id === btn.dataset.id; });
      if(sh){
        sh.isPadel = !sh.isPadel;
        addHistory('"'+sh.name+'" marcado como '+(sh.isPadel?'Padel':'Uso diário'));
        saveState(); render();
      }
      return;
    }
    if(act === 'shoe-move'){
      const sh = state.shoes.find(function(x){ return x.id === btn.dataset.id; });
      if(sh){
        sh.location = btn.dataset.loc;
        addHistory('Calçado "'+sh.name+'" → '+(sh.location==='lisboa'?'Lisboa':'Coimbra'));
        saveState(); render(); showToast('Calçado movido');
      }
      return;
    }

    // Mala
    if(act === 'mala-dir'){
      ui.malaDir = btn.dataset.dir;
      if(ui.malaDir === 'l2c') initL2CDefaults();
      else resetC2LSelections();
      render(); return;
    }
    if(act === 'fill-c2l-uw'){
      const a = calcAutonomy();
      ui.c2lBoxers = Math.max(0, state.goals.boxers - a.boxers);
      ui.c2lMeias = Math.max(0, state.goals.meias - a.meias);
      render(); showToast('Défice interior preenchido'); return;
    }
    if(act === 'confirm-l2c'){
      const bx = Math.min(state.underwear.boxers.vermelho, num(ui.l2cBoxers, 0));
      const me = Math.min(state.underwear.meias.vermelho, num(ui.l2cMeias, 0));
      let movedItems = 0, movedPij = 0, movedShoes = 0;

      state.underwear.boxers.vermelho = Math.max(0, state.underwear.boxers.vermelho - bx);
      state.underwear.meias.vermelho = Math.max(0, state.underwear.meias.vermelho - me);

      state.items.forEach(function(it){
        if(it.location === 'lisboa' && ui.l2cItems[it.id]){
          it.location = 'casa';
          movedItems++;
        }
      });
      PIJAMA_DEFS.forEach(function(d){
        if(state.pijama[d.key] && state.pijama[d.key].inLisboa && ui.l2cPijama[d.key]){
          state.pijama[d.key] = { inLisboa:false, status:'verde', reasons:[], uses:0 };
          movedPij++;
        }
      });
      state.shoes.forEach(function(s){
        if(s.location === 'lisboa' && ui.l2cShoes[s.id]){
          s.location = 'casa';
          movedShoes++;
        }
      });

      if(bx + me + movedItems + movedPij + movedShoes === 0){
        showToast('Nada selecionado na mala.'); return;
      }
      addHistory('Mala Lisboa→Coimbra: '+movedItems+' peça(s), '+movedPij+' pijama, '+movedShoes+' calçado, '+bx+'B/'+me+'M');
      initL2CDefaults();
      saveState(); render(); showToast('Mala Lisboa → Coimbra confirmada!'); return;
    }
    if(act === 'confirm-c2l'){
      const bx = num(ui.c2lBoxers, 0);
      const me = num(ui.c2lMeias, 0);
      let movedItems = 0, movedPij = 0, movedShoes = 0;

      state.underwear.boxers.verde += bx;
      state.underwear.meias.verde += me;

      state.items.forEach(function(it){
        if(it.location === 'casa' && ui.c2lItems[it.id]){
          it.location = 'lisboa';
          it.status = 'verde';
          it.reasons = [];
          it.uses = 0;
          movedItems++;
        }
      });
      PIJAMA_DEFS.forEach(function(d){
        if(ui.c2lPijama[d.key]){
          state.pijama[d.key] = { inLisboa:true, status:'verde', reasons:[], uses:0 };
          movedPij++;
        }
      });
      state.shoes.forEach(function(s){
        if(s.location === 'casa' && ui.c2lShoes[s.id]){
          s.location = 'lisboa';
          movedShoes++;
        }
      });

      if(bx + me + movedItems + movedPij + movedShoes === 0){
        showToast('Nada selecionado na mala.'); return;
      }
      addHistory('Mala Coimbra→Lisboa: '+movedItems+' peça(s), '+movedPij+' pijama, '+movedShoes+' calçado, +'+bx+'B/+'+me+'M');
      resetC2LSelections();
      ui.l2cInitialized = false;
      saveState(); render(); showToast('Mala Coimbra → Lisboa confirmada!'); return;
    }

    // Sync & Metas
    if(act === 'copy-md'){ copyWithFallback(buildMarkdownSummary(), 'Resumo Markdown'); return; }
    if(act === 'export-json'){ copyWithFallback(JSON.stringify(state, null, 2), 'Backup JSON'); return; }
    if(act === 'download-json'){
      try {
        const blob = new Blob([JSON.stringify(state, null, 2)], { type:'application/json' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'guarda-roupa-lisboa-backup.json';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        setTimeout(function(){ URL.revokeObjectURL(url); }, 1000);
        showToast('Backup .json descarregado!');
      } catch(_) {
        openCopyModal(JSON.stringify(state, null, 2), 'Backup JSON');
      }
      return;
    }
    if(act === 'open-import'){ openImportModal(); return; }
    if(act === 'do-import'){
      const ta = document.getElementById('import-text');
      const txt = ta ? ta.value.trim() : '';
      if(!txt){ showToast('Cola o JSON ou escolhe um ficheiro.'); return; }
      try {
        state = migrateState(JSON.parse(txt));
        addHistory('Backup JSON importado');
        ui.l2cInitialized = false;
        resetC2LSelections();
        saveState(); closeModal(); render(); showToast('Dados importados com sucesso!');
      } catch(err) {
        showToast('JSON inválido: ' + err.message);
      }
      return;
    }
    if(act === 'reset-goals'){
      state.goals = defaultGoals();
      addHistory('Metas repostas para padrão 4 dias (Seg–Qui)');
      saveState(); render(); showToast('Metas repostas (4 dias)'); return;
    }
    if(act === 'ask-sample'){ openConfirmModal('Carregar exemplos', 'Isto substitui os dados atuais por exemplos de demonstração.', 'do-sample'); return; }
    if(act === 'do-sample'){
      state = sampleState();
      ui.l2cInitialized = false; resetC2LSelections();
      saveState(); closeModal(); render(); showToast('Exemplos carregados'); return;
    }
    if(act === 'ask-clear'){ openConfirmModal('Limpar tudo', 'Isto apaga todas as peças, calçado e contagens. Queres continuar?', 'do-clear'); return; }
    if(act === 'do-clear'){
      state = emptyState();
      addHistory('Inventário limpo');
      ui.l2cInitialized = false; resetC2LSelections();
      saveState(); closeModal(); render(); showToast('Inventário limpo'); return;
    }
  });

  // Event delegation: input & change
  document.addEventListener('input', function(e){
    const t = e.target;
    if(!t) return;
    if(t.id === 'search-input'){
      ui.search = t.value;
      const pos = t.selectionStart;
      render();
      const el = document.getElementById('search-input');
      if(el){ el.focus(); try { el.setSelectionRange(pos, pos); } catch(_){} }
      return;
    }
    if(t.id === 's-name' && !document.getElementById('s-id').value){
      const chk = document.getElementById('s-padel');
      if(chk) chk.checked = /padel/i.test(t.value);
      return;
    }
    if(t.id === 'l2c-boxers'){ ui.l2cBoxers = num(t.value, 0); return; }
    if(t.id === 'l2c-meias'){ ui.l2cMeias = num(t.value, 0); return; }
    if(t.id === 'c2l-boxers'){ ui.c2lBoxers = num(t.value, 0); return; }
    if(t.id === 'c2l-meias'){ ui.c2lMeias = num(t.value, 0); return; }
  });

  document.addEventListener('change', function(e){
    const t = e.target;
    if(!t) return;
    if(t.id === 'cat-filter'){ ui.catFilter = t.value; render(); return; }
    if(t.id === 'import-file' && t.files && t.files[0]){
      const reader = new FileReader();
      reader.onload = function(){
        const ta = document.getElementById('import-text');
        if(ta) ta.value = String(reader.result || '');
      };
      reader.readAsText(t.files[0]);
      return;
    }
    const act = t.dataset.action;
    if(act === 'chk-l2c-item'){ ui.l2cItems[t.dataset.id] = !!t.checked; return; }
    if(act === 'chk-l2c-pij'){ ui.l2cPijama[t.dataset.key] = !!t.checked; return; }
    if(act === 'chk-l2c-shoe'){ ui.l2cShoes[t.dataset.id] = !!t.checked; return; }
    if(act === 'chk-c2l-item'){ ui.c2lItems[t.dataset.id] = !!t.checked; return; }
    if(act === 'chk-c2l-pij'){ ui.c2lPijama[t.dataset.key] = !!t.checked; return; }
    if(act === 'chk-c2l-shoe'){ ui.c2lShoes[t.dataset.id] = !!t.checked; return; }
  });

  // Event delegation: form submit
  document.addEventListener('submit', function(e){
    const form = e.target.closest('form[data-action]');
    if(!form) return;
    e.preventDefault();
    const act = form.dataset.action;

    if(act === 'submit-item'){
      const id = document.getElementById('f-id').value;
      const name = document.getElementById('f-name').value.trim();
      const category = document.getElementById('f-cat').value;
      const location = document.getElementById('f-loc').value;
      const note = document.getElementById('f-note').value.trim();
      if(!name){ showToast('Indica o nome da peça'); return; }
      const st = ui.modalStatus;
      const reasons = st === 'verde' ? [] : ui.modalReasons.slice();

      if(id){
        const ex = state.items.find(function(x){ return x.id === id; });
        if(ex){
          ex.name = name; ex.category = category; ex.location = location;
          ex.status = st; ex.reasons = reasons; ex.note = note;
          addHistory('Peça editada: '+name);
        }
      } else {
        state.items.push({ id:uid('it'), name:name, category:category, location:location, status:st, reasons:reasons, uses:0, note:note });
        addHistory('Peça adicionada: '+name+' ('+category+')');
      }
      ui.l2cInitialized = false;
      saveState(); closeModal(); render(); showToast(id ? 'Peça atualizada' : 'Peça adicionada');
      return;
    }

    if(act === 'submit-shoe'){
      const id = document.getElementById('s-id').value;
      const name = document.getElementById('s-name').value.trim();
      const location = document.getElementById('s-loc').value;
      const isPadel = !!document.getElementById('s-padel').checked;
      const note = document.getElementById('s-note').value.trim();
      if(!name){ showToast('Indica o nome do calçado'); return; }

      if(id){
        const ex = state.shoes.find(function(x){ return x.id === id; });
        if(ex){
          ex.name = name; ex.location = location; ex.isPadel = isPadel; ex.note = note;
          addHistory('Calçado editado: '+name);
        }
      } else {
        state.shoes.push({ id:uid('sh'), name:name, location:location, isPadel:isPadel, note:note });
        addHistory('Calçado adicionado: '+name);
      }
      saveState(); closeModal(); render(); showToast(id ? 'Calçado atualizado' : 'Calçado adicionado');
      return;
    }

    if(act === 'save-goals'){
      const g = {
        days: Math.max(1, num(document.getElementById('g-days').value, 4)),
        boxers: Math.max(1, num(document.getElementById('g-boxers').value, 4)),
        meias: Math.max(1, num(document.getElementById('g-meias').value, 4)),
        tops: Math.max(1, num(document.getElementById('g-tops').value, 4)),
        calcas: Math.max(1, num(document.getElementById('g-calcas').value, 2)),
        camisolasMin: Math.max(1, num(document.getElementById('g-cam-min').value, 1)),
        camisolasIdeal: Math.max(1, num(document.getElementById('g-cam-idl').value, 3)),
        casacosMin: Math.max(1, num(document.getElementById('g-cas-min').value, 1)),
        casacosIdeal: Math.max(1, num(document.getElementById('g-cas-idl').value, 2)),
        sapatilhasMin: Math.max(1, num(document.getElementById('g-sh-min').value, 1)),
        sapatilhasIdeal: Math.max(1, num(document.getElementById('g-sh-idl').value, 2))
      };
      if(g.camisolasIdeal < g.camisolasMin) g.camisolasIdeal = g.camisolasMin;
      if(g.casacosIdeal < g.casacosMin) g.casacosIdeal = g.casacosMin;
      if(g.sapatilhasIdeal < g.sapatilhasMin) g.sapatilhasIdeal = g.sapatilhasMin;
      state.goals = g;
      addHistory('Metas de autonomia atualizadas');
      saveState(); render(); showToast('Metas guardadas!');
      return;
    }
  });

  render();
})();
</script>
</body>
</html>
